def recv_until(sock, marker=b"\r\n\r\n"):
    data = b""
    while marker not in data:
        data += sock.recv(4096)
        if len(data) > 65536:
            raise RuntimeError("header too large")
    return data.split(marker, 1)
def ws_frame(sock, stash):
    def need(n):
        nonlocal stash
        while len(stash) < n:
            chunk = sock.recv(max(4096, n-len(stash)))
            if not chunk: raise EOFError()
            stash += chunk
        out, stash = stash[:n], stash[n:]
        return out
    a,b = need(2)
    n = b & 127
    if n == 126: n = struct.unpack("!H", need(2))[0]
    if n == 127: n = struct.unpack("!Q", need(8))[0]
    payload = need(n)
    return (a & 15, payload), stash
def ws_send_text(sock, text):
    data = text.encode()
    mask = os.urandom(4)
    if len(data) < 126: hdr = bytes([0x81, 0x80 | len(data)])
    else: hdr = bytes([0x81, 0x80 | 126]) + struct.pack("!H", len(data))
    sock.sendall(hdr + mask + bytes(v ^ mask[i % 4] for i, v in enumerate(data)))
